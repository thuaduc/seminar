\begin{frame}{What is an autonomous coding agent?}
  A system that, given a natural-language request, \textbf{autonomously} produces a
  working code change --- navigating files, editing, executing, and revising
  without step-by-step human guidance.

  \bigskip
  \begin{itemize}
    \item \textbf{Autonomous} --- decides what to do next, not told
      \medskip
    \item \textbf{Grounded in execution} --- runs tests, reads errors, acts on
      them
      \medskip
    \item \textbf{Repository-scale} --- works across files in a real codebase,
      not on isolated functions
  \end{itemize}
\end{frame}

\begin{frame}{How the loop is controlled}
  The five stages say \textbf{what} an agent does. Two more questions decide
  \textbf{how}:

  \bigskip
  \begin{itemize}
    \item \textbf{Orchestration} --- who decides what happens next?
      The model, a hard-coded script, or a search algorithm?
      \medskip
    \item \textbf{Interface \& context} --- what can the agent see and do?
      A purpose-built command set, or a raw sandbox?
  \end{itemize}

  \bigskip
  These are not stages. They cut \emph{across} every stage --- they decide
  whether the loop runs at all, and what it has access to.
\end{frame}

% THE RECONCILIATION. D4C and Agentless look like a contradiction -- one says

% resolves it by scale, and that resolution is this frame's only job.
\begin{frame}{Plan: the function boundary}
  D4C reports that \textbf{perfect fault localisation makes repair
  worse}~\cite{xu2024d4c}. Frames~\ref{fr:planning}--12 said localisation is the
  problem. Both are right.

  \bigskip
  \begin{itemize}
    \item \textbf{Above} the function boundary --- \emph{which} function, among
      tens of thousands? No pre-training prior answers that. Localisation
      \textbf{is} the task.
      \medskip
    \item \textbf{Below} it --- ranking \emph{lines within} a function already
      known to be faulty fights the model's infilling prior.
  \end{itemize}

  \bigskip
  {\small D4C repairs whole functions: \textbf{180 of 437} single-function
  Defects4J bugs, beating systems given perfect fault localisation by
  10\,\%~\cite{xu2024d4c}.}
\end{frame}

\begin{frame}{Plan: the Agentless paradox}
  \label{fr:planlast}
  \begin{itemize}
    \item \textbf{Plan-then-generate} --- inspectable, brittle if the plan is
      wrong~\cite{xia2024agentless}
    \item \textbf{Interleaved} --- adaptive, but spread over many turns and hard
      to audit~\cite{yao2023react}
    \item \textbf{Generate-then-refine} --- first output as
      hypothesis~\cite{madaan2023selfrefine}
  \end{itemize}

  \bigskip
  \begin{block}{The system that plans \emph{least} is competitive with those that
    plan most}
    So the claim is not that deliberation helps. It is that
    \textbf{grounding} helps --- and Agentless gets its grounding
    \emph{structurally}, hard-coding the narrowing an agentic loop has to
    rediscover by exploring.
  \end{block}
\end{frame}

\begin{frame}{Generate: writing the patch}
  The model writes or edits source code --- a patch, a new function, a refactored
  block.

  \medskip
  \begin{itemize}
    \item \textbf{Infilling} --- the model completes a span given surrounding
      context. Works well within a single function.
      \medskip
    \item \textbf{Diff generation} --- produce a unified diff rather than a full
      file. Smaller output, but fragile if the context is stale.
      \medskip
    \item \textbf{Sample $N$, pick one} --- generate many candidates, rank by
      test results or a critic. AlphaCode samples up to \emph{millions} and
      clusters by execution~\cite{li2022alphacode}.
  \end{itemize}

  \medskip
  \begin{block}{Two correct patches can share no token}
    There is no canonical form --- string-similarity metrics fail. Ranking
    must use \textbf{execution} or \textbf{AST normalisation}, not text
    comparison.
  \end{block}
\end{frame}

\begin{frame}{Execute: running the code}
  The agent runs what it wrote. This is where the \textbf{environment can say no}.

  \medskip
  \begin{itemize}
    \item \textbf{Test suite} --- fail-to-pass tests must flip, pass-to-pass
      must not regress.
      \medskip
    \item \textbf{Linter / type checker} --- catches syntax errors and type
      mismatches before tests run.
      \medskip
    \item \textbf{Error traces} --- stack traces and assertion messages tell the
      agent \emph{what} failed and \emph{where}.
  \end{itemize}

  \medskip
  \begin{block}{The test suite is a sample, not the spec}
    A patch can pass every test and still be wrong. The grading oracle is
    \textbf{incomplete} --- it checks what the tests exercise, nothing more.
  \end{block}
\end{frame}

\begin{frame}{SWE-bench}
  \label{fr:swebench}
  Merged pull requests that \textbf{close an issue} and \textbf{touch the test
  suite}~\cite{jimenez2024swebench}.

  \bigskip
  \begin{enumerate}
    \item \textbf{Given} --- the issue text $+$ the repository \emph{before}
      the merge.
    \item \textbf{Produce} --- a patch.
    \item \textbf{Graded} --- \emph{fail-to-pass} must flip,
      \emph{pass-to-pass} must not regress.
  \end{enumerate}

  \bigskip
  \begin{block}{The two filters do different work}
    Tests give the \textbf{grading oracle}; the issue link gives the
    \textbf{intent}.
  \end{block}

  \medskip
  {\small \textbf{Resolve rate} $=$ both test sets pass, first patch (pass@1).}
\end{frame}

% ------------------------------------------------------------- frame 15 ---
% "How to assess the quality of generated code" -- the flatly missing part in the
% old deck. Order: what we cannot use, what we use instead, what it costs.
\begin{frame}{Measuring generated code}
  \textbf{Not} string similarity --- two correct patches may share no token.

  \medskip
  \begin{itemize}
    \item \textbf{Functional correctness} --- run it. The only kind that
      survives.
      \smallskip
    \item \textbf{pass@$k$} --- one of $k$ samples passes; unbiased estimator
      over a larger pool~\cite{chen2021humaneval}.
      \smallskip
    \item \textbf{Resolve rate} --- pass@1 at repository scale
      (frame~\ref{fr:swebench}).
      \smallskip
    \item \textbf{Cost} --- USD, calls, wall-clock. It \emph{decouples} from
      resolve rate~\cite{sweeffi2025}.
  \end{itemize}

  \medskip
  \begin{block}{What none of them measure}
    All score the \textbf{end} of the pipeline. Plan quality, localisation,
    refinement efficiency: unmeasured.
  \end{block}
\end{frame}

        & 42.2\,\% \\
      \bottomrule
    \end{tabular}
  \end{center}

  \smallskip
  \begin{itemize}
    \item Easy $+$ medium are \textbf{91\,\% of Verified}. The headline mostly
      reports sub-hour work.
    \item The hard tier \textbf{stopped} near 20\,\%. A system can gain ten
      points of \emph{overall} resolve rate while solving no new hard task.
    \item Easy and medium are an \textbf{ensembling} problem --- different
      systems already solve different instances, so pooling them wins. Hard is
      not: over half of it defeats every system at once.
  \end{itemize}
\end{frame}

\begin{frame}{Challenge: intent capture}
  \label{fr:ambiguity}
  The stage with \emph{no feedback edge} (frame~\ref{fr:stages}). The agent
  \textbf{guesses and commits}.

  \medskip
  \begin{itemize}
    \item Issue reports leave out edge cases and implicit invariants.
      \medskip
    \item Tests constrain only what they exercise. A wrong guess about an
      \textbf{untested invariant} scores as a success.
      \medskip
    \item \textbf{No surveyed system asks a clarifying question.}
  \end{itemize}

  \medskip
  \begin{block}{The gap is the evaluation, not the mechanism}
    Clarification mechanisms exist and
    work~\cite{lahiri2022ticoder,mu2024clarifygpt,vijayvargiya2026ambiguity}.
    SWE-bench filters out ambiguity by construction --- the agent is never
    rewarded for asking.
  \end{block}
\end{frame}

\begin{frame}{Challenge: plan}
  The repository does not fit in context. The agent must \textbf{choose what to
  read}, and that choice is most of the problem.

  \medskip
  \begin{itemize}
    \item Hard tier stuck at \textbf{20\,\%} against easy \textbf{80\,\%}
      (frame~\ref{fr:numbers})~\cite{ganhotra2025difficulty}.
      \medskip
    \item Of the 45 hard instances, 25 need several files --- only \textbf{9}
      are solved by anyone at all.
      \medskip
    \item Is the cause \textbf{retrieval} (never sees the right files) or
      \textbf{reasoning} (cannot compose the change)?
  \end{itemize}

  \medskip
  \begin{block}{The cheapest experiment nobody has run}
    Hand it the \textbf{gold file set}. Hard tier moves $\Rightarrow$ retrieval.
    It does not $\Rightarrow$ only training
    will~\cite{wei2025swerl,pan2024swegym}.
  \end{block}
\end{frame}

\begin{frame}{Challenge: generation}
  Two correct patches can share \textbf{no token}. Two near-identical patches
  can differ in meaning.

  \medskip
  \begin{itemize}
    \item You cannot score generated code against a reference --- every
      string-similarity metric fails.
      \medskip
    \item The dominant quality lever --- \textbf{sample $N$, rerank} ---
      multiplies cost directly~\cite{turncontrol2025}.
      \medskip
    \item Resolve rate \textbf{decouples from compute} --- spending more does
      not order the systems~\cite{sweeffi2025}.
  \end{itemize}

  \medskip
  \begin{block}{}
    \centering
    The practical frontier is not peak accuracy.\\
    It is \textbf{accuracy per dollar} and \textbf{per minute}.
  \end{block}
\end{frame}

\begin{frame}{Challenge: execution}
  The test suite is a \textbf{sample} of the intended behaviour, not the
  behaviour itself.

  \medskip
  \begin{itemize}
    \item A patch can pass every test and \textbf{still be wrong}.
      \medskip
    \item Reward hacking: closing test-leakage channels cuts scores by
      \textbf{double digits} --- agents partly learn the
      grader~\cite{cursor2025rewardhacking}.
      \medskip
    \item Contamination: models reproduce gold patches from the task ID
      alone~\cite{openai2026swebench}. OpenAI dropped Verified in 2026.
  \end{itemize}

  \medskip
  \begin{block}{The oracle decides when the loop stops}
    ``All tests pass'' smuggles ground truth into the stopping condition ---
    any self-critique inside that loop will look more effective than it is.
  \end{block}
\end{frame}

\begin{frame}{Challenge: refinement}
  Real fixes touch \textbf{several files} coherently, over a long sequence of
  steps.

  \medskip
  \begin{itemize}
    \item The marginal value of another repair round drops fast. At fixed
      budget, another round is often \emph{worse} than a fresh
      sample~\cite{olausson2024selfrepair}.
      \medskip
    \item Self-critique alone can only reorganise what the model already
      believes --- measured going
      \textbf{negative}~\cite{huang2024selfcorrect}.
      \medskip
    \item Pro shows the same ceiling as the hard tier: long-horizon,
      multi-file tasks do not yield to more
      iterations~\cite{deng2025swebenchpro}.
  \end{itemize}

  \medskip
  \begin{block}{Iterative repair vs.\ best-of-$N$: the same lever}
    Extra inference to cut variance, pulled sequentially or in parallel. The
    question is which allocation of the same tokens buys more correctness.
  \end{block}
\end{frame}

\begin{frame}{What it adds up to}
  Everything that works puts \textbf{something outside the model} in a position to
  contradict it. Sorted by \emph{what closes the loop}:

  \smallskip
  \scriptsize
  \begin{tabular}{@{}L{0.30\textwidth}L{0.64\textwidth}@{}}
    \toprule
    \textbf{Oracle} \newline {\scriptsize can surprise the model}
      & Self-debugging \emph{with tests} \quad D4C \quad Agentless validate \\
    \textbf{Proxy} \newline {\scriptsize carries information, nothing new enters}

  \end{tabular}

  \normalsize
  \begin{block}{So the two weak stages are the two with no oracle}
    \textbf{Intent} --- the oracle is a human, nobody asks. \quad
    \textbf{Validation} --- a suite a wrong patch passes.
  \end{block}
\end{frame}

\begin{frame}{Takeaways}
  \begin{enumerate}
    \item \textbf{Agentic beats one-shot because it can run things}, not
      because it reasons more.
      \medskip
    \item \textbf{One number no longer suffices.} Difficulty, contamination and
      cost belong next to any resolve rate.
      \medskip
    \item \textbf{Verification is the bottleneck.} In a randomised trial,
      developers were 19\,\% \emph{slower} --- believing they were 20\,\%
      faster~\cite{metr2025rct}.
  \end{enumerate}

  \bigskip
  \begin{block}{}
    \centering
    Autocomplete to autonomous issue resolution: crossed.\\
    Impressive to \textbf{dependable} is a question about oracles.
  \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Backup: the context bottleneck}
  \begin{itemize}
    \item Longer windows do not settle \emph{what to read}. Keeping full
      dependency context works best; padding with loosely related code
      \textbf{hurts} correctness~\cite{repoexec2025}.
      \medskip
    \item Appending every observation causes context explosion and semantic
      drift: early critical information gets buried~\cite{contextastool2025}.
      \medskip
    \item The whole prefix is re-sent each turn, so tokens over $n$ turns grow as
      $O(n^2)$~\cite{turncontrol2025}. A protocol property, not a finding;
      caching cuts the constant, not the growth.
  \end{itemize}

  \bigskip
  {\small An agent cannot fix what it never places in context.}
\end{frame}
